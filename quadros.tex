%===== PARA USAR QUADROS (versão ajustada e mais robusta) =====
\usepackage{float}      % para criar novos ambientes flutuantes
\usepackage{caption}    % para configurar o caption

% ===== AMBIENTE QUADRO =====

\newfloat{quadro}{htbp}{loq}
\floatname{quadro}{Quadro}

% Numeração simples
\renewcommand{\thequadro}{\arabic{quadro}}

% Lista de Quadros
\providecommand{\listquadroname}{Lista de Quadros}
\newcommand{\listofquadro}{\listof{quadro}{\listquadroname}}

% Formato do caption
\DeclareCaptionLabelFormat{quadronum}{#1~#2\ - }

\captionsetup[quadro]{
    name=Quadro,
    labelformat=quadronum,
    labelsep=none,
    position=top,
    font=normal,
    labelfont=bf
}
